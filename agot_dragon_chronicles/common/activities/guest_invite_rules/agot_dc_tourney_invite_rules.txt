# AGOT - Dragon Chronicles : règle d'invitation du Tournoi aérien
# Cavaliers de dragon (dragon vivant) parmi les vassaux, la cour et la famille proche.

agot_dc_invite_rule_dragonriders = {
	effect = {
		every_vassal_or_below = {
			limit = { agot_dc_rider_with_dragon_trigger = yes }
			add_to_list = characters
		}
		every_courtier_or_guest = {
			limit = { agot_dc_rider_with_dragon_trigger = yes }
			add_to_list = characters
		}
		every_close_family_member = {
			limit = {
				agot_dc_rider_with_dragon_trigger = yes
				NOT = { is_in_list = characters }
			}
			add_to_list = characters
		}
		liege ?= {
			if = {
				limit = {
					agot_dc_rider_with_dragon_trigger = yes
					NOT = { this = root }
				}
				add_to_list = characters
			}
		}
	}
}
