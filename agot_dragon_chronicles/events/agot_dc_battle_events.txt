namespace = agot_dc_battle

###############################################################################
# AGOT - Dragon Chronicles : évènements aléatoires de bataille
# 0100 / 0200 sont des gestionnaires cachés (scope = combat_side) appelés depuis
# agot_dc_combat_winner / agot_dc_combat_loser. Ils cherchent un cavalier de dragon
# dans le camp et lui envoient l'évènement visible.
#
# scope:dc_rider           = le cavalier
# scope:dc_dragon          = son dragon
# scope:dc_enemy_commander = le commandant adverse (peut ne pas exister)
###############################################################################

scripted_trigger agot_dc_side_has_rider_trigger = {
	OR = {
		side_commander ?= { agot_dc_battle_rider_trigger = yes }
		any_side_knight = { agot_dc_battle_rider_trigger = yes }
	}
}

scripted_effect agot_dc_save_battle_scopes_effect = {
	if = {
		limit = { side_commander ?= { agot_dc_battle_rider_trigger = yes } }
		side_commander = { save_scope_as = dc_rider }
	}
	else = {
		random_side_knight = {
			limit = { agot_dc_battle_rider_trigger = yes }
			save_scope_as = dc_rider
		}
	}
	scope:dc_rider.var:current_dragon = { save_scope_as = dc_dragon }
	enemy_side.side_commander ?= {
		if = {
			limit = { is_alive = yes }
			save_scope_as = dc_enemy_commander
		}
	}
}

#########################################
# 0100 - Gestionnaire : victoire
#########################################
agot_dc_battle.0100 = {
	hidden = yes
	scope = combat_side
	trigger = { agot_dc_side_has_rider_trigger = yes }
	immediate = {
		agot_dc_save_battle_scopes_effect = yes
		scope:dc_rider = {
			random_list = {
				10 = { trigger_event = agot_dc_battle.1001 } # Massacre des fuyards
				10 = { trigger_event = agot_dc_battle.1011 } # Carreau de scorpion
				10 = { # Le commandant ennemi entre les crocs
					trigger = {
						exists = scope:dc_enemy_commander
						scope:dc_enemy_commander = {
							NOT = { has_trait = dragonrider }
							NOT = { has_trait = dragon }
						}
					}
					trigger_event = agot_dc_battle.1021
				}
				8 = { trigger_event = agot_dc_battle.1031 } # Le rugissement de la victoire
			}
		}
	}
}

# 1001 - Massacre des fuyards
#########################################
agot_dc_battle.1001 = {
	type = character_event
	title = agot_dc_battle.1001.t
	desc = agot_dc_battle.1001.desc
	theme = battle
	left_portrait = {
		character = root
		animation = jockey_gallop
		camera = camera_dragonrider_over_battlefield
	}

	option = { # Brûler les fuyards
		name = agot_dc_battle.1001.a
		add_dread = 15
		add_prestige = medium_prestige_gain
		add_character_modifier = { modifier = agot_dc_dragon_terror_modifier years = 3 }

		stress_impact = {
			compassionate = major_stress_impact_gain
			forgiving = medium_stress_impact_gain
		}

			base = 50
			modifier = { add = 40 has_trait = wrathful }
		}
	}
	option = { # Les épargner
		name = agot_dc_battle.1001.b
		add_prestige = minor_prestige_gain
		add_character_modifier = { modifier = agot_dc_dragon_merciful_modifier years = 5 }
		stress_impact = {
			sadistic = medium_stress_impact_gain
			wrathful = minor_stress_impact_gain
		}
		ai_chance = {
			base = 50
			modifier = { add = 40 has_trait = compassionate }
		}
	}
}
