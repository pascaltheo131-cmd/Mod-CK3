#########################################
# 1081 - Enlèvement en plein vol (victoire)
#########################################
agot_dc_battle.1081 = {
	type = character_event
	title = agot_dc_battle.1081.t
	desc = agot_dc_battle.1081.desc
	theme = battle
	left_portrait = { character = root animation = personality_bold }
	right_portrait = { character = scope:dc_enemy_commander animation = fear }
	lower_right_portrait = scope:dc_dragon
	trigger = { scope:dc_enemy_commander ?= { is_alive = yes } }

	option = { # Exiger une rançon
		name = agot_dc_battle.1081.a
		scope:dc_enemy_commander = {
			pay_short_term_gold = { target = root gold = medium_gold_value }
			add_opinion = { target = root modifier = angry_opinion opinion = -20 }
		}
		add_prestige = medium_prestige_gain
		ai_chance = { base = 60 }
	}
	option = { # Le relâcher en plein ciel… au-dessus de ses lignes
		name = agot_dc_battle.1081.b
		add_dread = 20
		scope:dc_enemy_commander = {
			increase_wounds_effect = { REASON = dragon }
			add_opinion = { target = root modifier = scared_opinion opinion = -40 }
		}
		ai_chance = {
			base = 40
			modifier = { add = 30 has_trait = sadistic }
		}
	}
}
