namespace = agot_dc

###############################################################################
# AGOT - Dragon Chronicles : évènements aléatoires de temps de paix
# Envoyés depuis agot_dc_yearly_dragon_pulse au cavalier (root).
# scope:dc_dragon = le dragon du cavalier
###############################################################################

scripted_effect agot_dc_save_dragon_effect = {
	var:current_dragon = { save_scope_as = dc_dragon }
}

#########################################
# 0001 - La viande empoisonnée
# Quelqu'un a tenté d'empoisonner le dragon
#########################################
agot_dc.0001 = {
	type = character_event
	title = agot_dc.0001.t
	desc = agot_dc.0001.desc
	theme = intrigue
	cooldown = { years = 15 }
	left_portrait = { character = root animation = anger }
	right_portrait = {
		character = scope:dc_dragon
		animation = dragon_main
		camera = camera_dragon_event_standing
		outfit_tags = { linear_camera_zoom }
	}
	trigger = {
		agot_dc_rider_with_dragon_trigger = yes
		is_landed = yes
		var:current_dragon = { NOT = { has_trait = dragon_ill } }
	}
	immediate = { agot_dc_save_dragon_effect = yes }

	option = { # Mener l'enquête
		name = agot_dc.0001.a
		duel = {
			skill = intrigue
			value = 12
			50 = {
				desc = agot_dc.0001.a.found
				compare_modifier = { value = scope:duel_value multiplier = 3.5 }
				add_dread = 15
				add_prestige = medium_prestige_gain
				add_intrigue_lifestyle_xp = medium_lifestyle_xp
			}
			50 = {
				desc = agot_dc.0001.a.lost
				compare_modifier = { value = scope:duel_value multiplier = -3.5 }
				add_stress = medium_stress_gain
			}
		}
		ai_chance = {
			base = 50
			modifier = { add = 30 intrigue >= 12 }
		}
	}
	option = { # Faire goûter chaque repas par un prisonnier
		name = agot_dc.0001.b
		remove_short_term_gold = minor_gold_value
		add_dread = 10
		stress_impact = { compassionate = medium_stress_impact_gain }
		ai_chance = {
			base = 30
			modifier = { add = 30 has_trait = paranoid }
		}
	}
	option = { # Le dragon a mangé une partie de la viande
		name = agot_dc.0001.c
		random_list = {
			60 = { desc = agot_dc.0001.c.fine }
			40 = {
				desc = agot_dc.0001.c.ill
				scope:dc_dragon = { add_trait = dragon_ill }
			}
		}
		ai_chance = { base = 20 }
	}
}

		name = agot_dc.0051.a
		add_gold = medium_gold_value
		agot_add_dragon_training_xp = { VALUE = 15 }
		add_martial_lifestyle_xp = medium_lifestyle_xp
		scope:dc_vassal = { add_opinion = { target = root modifier = grateful_opinion opinion = 30 } }
		random = {
			chance = 15
			send_interface_toast = {
				title = agot_dc.0051.a.wildfire
				left_icon = root
				add_stress = minor_stress_gain
				add_prestige = minor_prestige_loss
			}
		}

	}
