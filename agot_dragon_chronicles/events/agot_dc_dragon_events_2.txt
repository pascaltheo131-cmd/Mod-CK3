	}
	immediate = {

		random_spouse = {
			limit = {
				is_alive = yes
				NOT = { has_trait = dragonrider }
			}
			save_scope_as = dc_spouse
		}
	}

	option = { # Présenter patiemment l'un à l'autre
		name = agot_dc.0141.a
		random_list = {
			70 = {
				desc = agot_dc.0141.a.success
				scope:dc_spouse = { add_opinion = { target = root modifier = grateful_opinion opinion = 20 } }
				add_stress = minor_stress_loss
			}
			30 = {
				desc = agot_dc.0141.a.burned
				scope:dc_spouse = { increase_wounds_effect = { REASON = dragon } }
			}
		}
