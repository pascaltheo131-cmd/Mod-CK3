# Le personnage est-il un dragon ?
dre_is_dragon = {
	has_trait = dragon
}

# Le personnage chevauche-t-il un dragon vivant ?
dre_has_dragon = {
	exists = var:current_dragon
	var:current_dragon = { is_alive = yes }
}
