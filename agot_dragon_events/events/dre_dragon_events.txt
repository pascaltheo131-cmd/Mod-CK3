# 0001 - Le dragon refuse d'obéir
dragon_ev.0001 = {
	type = character_event
	title = dragon_ev.0001.t
	desc = dragon_ev.0001.desc
	theme = martial
	left_portrait = { character = root animation = anger }
	right_portrait = { character = scope:dragon }
	trigger = { dre_has_dragon = yes }
	cooldown = { years = 5 }
	immediate = { dre_save_dragon = yes }

	option = { # Imposer sa volonté
		name = dragon_ev.0001.a
		random_list = {
			50 = {
				modifier = { add = prowess }
				desc = dragon_ev.0001.a.success
				add_prestige = 150
				add_character_modifier = { modifier = dre_dragon_bond_modifier years = 5 }
			}
			50 = {
				desc = dragon_ev.0001.a.fail
				increase_wounds_effect = { REASON = fight }
			}
		}
	}
	option = { # Patience
		name = dragon_ev.0001.b
		add_stress = 15
		add_piety = 50
	}
}

# 0005 - Le dragon s'en prend à un courtisan
dragon_ev.0005 = {
	type = character_event
	title = dragon_ev.0005.t
	desc = dragon_ev.0005.desc
	theme = dread
	left_portrait = { character = root animation = shock }
	right_portrait = { character = scope:victim animation = fear }
	trigger = {
		dre_has_dragon = yes
		any_courtier = { is_adult = yes dre_is_dragon = no }
	}
	cooldown = { years = 10 }
	immediate = {
		dre_save_dragon = yes
		random_courtier = {
			limit = { is_adult = yes dre_is_dragon = no }
			save_scope_as = victim
		}
	}

	option = { # Intervenir
		name = dragon_ev.0005.a
		scope:victim = { add_opinion = { target = root modifier = grateful_opinion opinion = 30 } }
		random = { chance = 30 increase_wounds_effect = { REASON = fight } }
	}
	option = { # Laisser faire
		name = dragon_ev.0005.b
		add_dread = 25
		scope:victim = { death = { death_reason = death_dragon } }
	}
}
