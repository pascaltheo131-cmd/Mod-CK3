agot_sh_pol.0001 = {
	type = character_event
	title = agot_sh_pol.0001.t
	desc = agot_sh_pol.0001.desc
	theme = realm
	left_portrait = {
		character = root
		animation = anger
	}
	right_portrait = {
		character = scope:agot_sh_pol_liege
		animation = dismissal
	}
	cooldown = { years = 5 }

	trigger = {
		agot_sh_is_active_vassal = yes
		OR = {
			agot_sh_is_ambitious = yes
			agot_sh_is_opportunist = yes
		}
		agot_sh_pol_holds_grievance_against_liege_trigger = yes
		is_a_faction_member = no
		liege = {
			any_targeting_faction = {
				root = { can_join_faction = prev }
			}
		}
	}

	immediate = {
		liege = {
			save_scope_as = agot_sh_pol_liege
			random_targeting_faction = {
				limit = {
					root = { can_join_faction = prev }
				}
				save_scope_as = agot_sh_pol_faction
			}
		}
	}

	option = { # Rejoindre ceux qui conspirent
		name = agot_sh_pol.0001.a
		join_faction = scope:agot_sh_pol_faction
		ai_chance = {
			base = 50
			modifier = { add = 30 agot_sh_is_ambitious = yes }
			modifier = { add = 20 has_trait = vengeful }
			modifier = { add = -30 has_trait = craven }
		}
	}
	option = { # Ruminer en silence
		name = agot_sh_pol.0001.b
		add_prestige = minor_prestige_loss
		ai_chance = {
			base = 30
			modifier = { add = 20 agot_sh_is_opportunist = yes }
			modifier = { add = 20 has_trait = patient }
		}
	}
	option = { # Pardonner
		name = agot_sh_pol.0001.c
		remove_variable = agot_sh_pol_grievance
		remove_opinion = {
			target = scope:agot_sh_pol_liege
			modifier = agot_sh_pol_grievance_opinion
		}
		stress_impact = {
			vengeful = major_stress_impact_gain
			wrathful = medium_stress_impact_gain
			forgiving = minor_stress_impact_loss
		}
		ai_chance = {
			base = 5
			modifier = { add = 30 has_trait = forgiving }
			modifier = { factor = 0 has_trait = vengeful }
		}
	}
}
