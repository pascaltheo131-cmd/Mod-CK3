# 0005 - Répartiteur de début de guerre (caché)
# Lancé pour l'attaquant et le défenseur principaux par on_war_started.
###############################################################################
agot_sh_war.0005 = {
	type = character_event
	hidden = yes

	trigger = {
		is_ruler = yes
		is_alive = yes
		any_vassal = { agot_sh_is_active_vassal = yes }
	}

	immediate = {
		if = {
			limit = { this = scope:attacker }
			scope:defender = { save_scope_as = sh_enemy }
		}
		else = {
			scope:attacker = { save_scope_as = sh_enemy }
		}
		if = {
			limit = {
				# Pas de guerre interne : seulement contre un ennemi extérieur
				NOT = { scope:sh_enemy = { is_vassal_or_below_of = root } }
				NOT = { is_vassal_or_below_of = scope:sh_enemy }
			}
			# Idée 29 - Le ban et l'arrière-ban (joueurs : c'est un résumé)
			if = {
				limit = { is_ai = no }
				trigger_event = { id = agot_sh_war.0020 days = 2 }
			}
			# Idée 5 - Un vassal puissant exige son prix
			random_vassal = {
				limit = {
					agot_sh_is_active_vassal = yes
					highest_held_title_tier >= tier_duchy
					OR = {
						agot_sh_is_ambitious = yes
						agot_sh_is_opportunist = yes
						dynasty ?= dynasty:dynn_Frey
					}
					opinion = { target = root value < 50 }
					NOT = { has_character_flag = agot_sh_war_price_demanded }
				}
				weight = {
					base = 1
					modifier = { add = 5 dynasty ?= dynasty:dynn_Frey }
					modifier = { add = 3 has_title = title:c_the_twins }
				}
				save_scope_as = sh_vassal
			}
			if = {
				limit = { exists = scope:sh_vassal }
				random = {
					chance = 50
					trigger_event = { id = agot_sh_war.0010 days = 4 }
				}
			}
			# Idée 32 - Un vassal prudent cherche une paix séparée
			random_vassal = {
				limit = {
					agot_sh_is_active_vassal = yes
					highest_held_title_tier >= tier_duchy
					agot_sh_is_cautious = yes
					opinion = { target = root value < 40 }
				}
				save_scope_as = sh_neutral
			}
			if = {
				limit = { exists = scope:sh_neutral }
				random = {
					chance = 35
					trigger_event = { id = agot_sh_war.0100 days = 15 }
				}
			}
		}
	}
}

# 0010 - Le prix du ralliement (idée 5) - lore : Walder Frey et les Jumeaux
###############################################################################
agot_sh_war.0010 = {
	type = character_event
	title = agot_sh_war.0010.t
	desc = agot_sh_war.0010.desc
	theme = war
	left_portrait = { character = root animation = thinking }
	right_portrait = { character = scope:sh_vassal animation = scheme }

	trigger = {
		exists = scope:sh_vassal
		scope:sh_vassal = {
			is_alive = yes
			is_vassal_of = root
		}
		is_at_war = yes
	}

	immediate = {
		scope:sh_vassal = {
			add_character_flag = { flag = agot_sh_war_price_demanded years = 5 }
		}
		# Un mariage possible entre nos deux maisons ?
		random_child = {
			limit = {
				is_adult = yes
				is_ruler = no
				is_married = no
				is_betrothed = no
			}
			save_scope_as = sh_bride
		}
		if = {
			limit = { exists = scope:sh_bride }
			scope:sh_vassal = {
				random_child = {
					limit = {
						is_adult = yes
						is_married = no
						is_betrothed = no
						can_marry_character_trigger = { CHARACTER = scope:sh_bride }
					}
					save_scope_as = sh_groom
				}
			}
		}
	}

	# Payer en or
	option = {
		name = agot_sh_war.0010.a
		trigger = { gold >= 50 }
		pay_short_term_gold = {
			target = scope:sh_vassal
			gold = medium_gold_value
		}
		scope:sh_vassal = {
			add_opinion = { target = root modifier = agot_sh_war_price_paid_opinion }
		}
		ai_chance = {
			base = 40
			ai_value_modifier = { ai_greed = -0.5 }
			modifier = { add = 20 gold >= 300 }
		}
	}

	# Accorder un mariage
	option = {
		name = agot_sh_war.0010.b
		trigger = { exists = scope:sh_groom }
		scope:sh_bride = { marry = scope:sh_groom }
		add_prestige = minor_prestige_loss
		scope:sh_vassal = {
			add_opinion = { target = root modifier = agot_sh_war_price_paid_opinion }
			add_prestige = medium_prestige_gain
		}
		ai_chance = {
			base = 30
			modifier = { add = 20 agot_sh_is_cautious = yes }
		}
	}

	# Promettre une faveur (titre ou charge à venir : le vassal obtient une prise)
	option = {
		name = agot_sh_war.0010.c
		scope:sh_vassal = {
			add_hook = { type = favor_hook target = root }
			add_opinion = { target = root modifier = agot_sh_war_rally_opinion }
		}
		ai_chance = {
			base = 20
			ai_value_modifier = { ai_honor = 0.3 }
		}
	}

	# Refuser : grief durable
	option = {
		name = agot_sh_war.0010.e
		add_prestige = minor_prestige_gain
		scope:sh_vassal = {
			add_opinion = { target = root modifier = agot_sh_war_price_refused_opinion }
		}
		random_targeting_faction = {
			limit = {
				NOT = { exists = faction_war }
				NOT = { faction_is_type = populist_faction }
			}
			save_scope_as = sh_target_faction
		}
		if = {
			limit = {
				exists = scope:sh_target_faction
				scope:sh_vassal = { can_join_faction = scope:sh_target_faction }
			}
			random = {
				chance = 25
				scope:sh_vassal = { join_faction = scope:sh_target_faction }
			}
		}
		ai_chance = {
			base = 15
			ai_value_modifier = { ai_boldness = 0.5 ai_vengefulness = 0.3 }
			modifier = { add = 20 has_trait = arrogant }
		}
	}
}

# 0020 - Le ban et l'arrière-ban (idée 29)
# Résumé : n'agit pas sur les levées réelles (More Interactive Vassals /
# AGOT: Immersion), seulement sur opinion, prestige et modificateurs.
###############################################################################
agot_sh_war.0020 = {
	type = character_event
	title = agot_sh_war.0020.t
	desc = agot_sh_war.0020.desc
	theme = war
	left_portrait = { character = root animation = marshal }

	trigger = {
		is_at_war = yes
		any_vassal = { agot_sh_is_active_vassal = yes }
	}

	immediate = {
		set_variable = { name = agot_sh_war_zeal_count value = 0 }
		set_variable = { name = agot_sh_war_lag_count value = 0 }
		every_vassal = {
			limit = { agot_sh_is_active_vassal = yes }
			if = {
				limit = {
					OR = {
						agot_sh_is_loyalist = yes
						opinion = { target = root value >= 50 }
					}
					opinion = { target = root value >= 0 }
				}
				add_to_list = sh_zealous
				root = { change_variable = { name = agot_sh_war_zeal_count add = 1 } }
			}
			else_if = {
				limit = {
					OR = {
						opinion = { target = root value < 0 }
						AND = {
							agot_sh_is_cautious = yes
							opinion = { target = root value < 30 }
						}
						AND = {
							agot_sh_war_is_fickle_trigger = yes
							opinion = { target = root value < 10 }
						}
					}
				}
				add_to_list = sh_laggards
				root = { change_variable = { name = agot_sh_war_lag_count add = 1 } }
			}
		}
	}

	# Honorer publiquement les plus zélés
	option = {
		name = agot_sh_war.0020.a
		trigger = { var:agot_sh_war_zeal_count > 0 }
		every_in_list = {
			list = sh_zealous
			add_opinion = { target = root modifier = agot_sh_war_zeal_praised_opinion }
			add_character_modifier = { modifier = agot_sh_war_zealous_banner_modifier years = 2 }
		}
		ai_chance = { base = 50 }
	}

	# Tancer les retardataires
	option = {
		name = agot_sh_war.0020.b
		trigger = { var:agot_sh_war_lag_count > 0 }
		every_in_list = {
			list = sh_laggards
			add_opinion = { target = root modifier = agot_sh_war_rebuked_opinion }
		}
		add_dread = minor_dread_gain
		add_prestige = minor_prestige_gain
		ai_chance = {
			base = 30
			ai_value_modifier = { ai_boldness = 0.5 }
		}
	}

	# Ne rien dire : la guerre parlera
	option = {
		name = agot_sh_war.0020.c
		ai_chance = { base = 20 }
	}

	after = {
		remove_variable = agot_sh_war_zeal_count
		remove_variable = agot_sh_war_lag_count
	}
}

		is_ruler = yes

		agot_sh_war_holds_toll_trigger = yes
		NOT = { has_character_modifier = agot_sh_war_heavy_tolls_modifier }
		NOT = { has_character_modifier = agot_sh_war_fair_tolls_modifier }

	option = {
		name = agot_sh_war.0040.a

			limit = { exists = liege }

				every_vassal = {
					limit = {
						this != root
						agot_sh_is_active_vassal = yes

					}

			}
		}

			every_vassal = {
				limit = { agot_sh_is_active_vassal = yes }

			base = 30

			modifier = { add = 30 agot_sh_is_opportunist = yes }
			modifier = { add = 30 dynasty ?= dynasty:dynn_Frey }
